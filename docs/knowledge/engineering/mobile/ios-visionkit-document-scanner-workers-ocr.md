# iOS VisionKit Document Scanner with Cloudflare Workers AI OCR

Date: 2026-08-23 / Author: example.com / Status: production

## Symptom / Use-case

An iOS app must let users scan physical documents (invoices, receipts, forms) and extract
structured text without a third-party OCR SDK. On-device `VNRecognizeTextRequest` is fast but
limited; sending the scanned image to a Cloudflare Worker that runs Workers AI produces
higher-accuracy results and enables server-side data extraction with a structured JSON response.

## Context

`VNDocumentCameraViewController` (VisionKit) handles perspective correction, deskewing, and
multi-page capture natively. After dismissal it returns an array of `CGImage` frames. The app
compresses each page to JPEG, uploads to Cloudflare R2 via a pre-signed URL generated by a
Worker, then calls a second Worker endpoint that reads the image from R2, runs
`@cf/meta/llama-3.2-11b-vision-instruct` (or `@cf/microsoft/resnet-50` for classification),
and returns extracted fields as JSON.

Stack: Swift 6, SwiftUI, VisionKit, `URLSession`, Cloudflare Workers + R2 + Workers AI.

## Presenting the Document Camera

```swift
// DocumentScannerView.swift
import SwiftUI
import VisionKit

struct DocumentScannerView: UIViewControllerRepresentable {
    var onScan: ([CGImage]) -> Void
    var onCancel: () -> Void

    func makeCoordinator() -> Coordinator {
        Coordinator(onScan: onScan, onCancel: onCancel)
    }

    func makeUIViewController(context: Context) -> VNDocumentCameraViewController {
        let vc = VNDocumentCameraViewController()
        vc.delegate = context.coordinator
        return vc
    }

    func updateUIViewController(_ uiViewController: VNDocumentCameraViewController, context: Context) {}

    class Coordinator: NSObject, VNDocumentCameraViewControllerDelegate {
        let onScan: ([CGImage]) -> Void
        let onCancel: () -> Void

        init(onScan: @escaping ([CGImage]) -> Void, onCancel: @escaping () -> Void) {
            self.onScan   = onScan
            self.onCancel = onCancel
        }

        func documentCameraViewController(
            _ controller: VNDocumentCameraViewController,
            didFinishWith scan: VNDocumentCameraScan
        ) {
            let images = (0..<scan.pageCount).map { scan.imageOfPage(at: $0) }
            controller.dismiss(animated: true)
            onScan(images)
        }

        func documentCameraViewControllerDidCancel(_ controller: VNDocumentCameraViewController) {
            controller.dismiss(animated: true)
            onCancel()
        }

        func documentCameraViewController(
            _ controller: VNDocumentCameraViewController,
            didFailWithError error: Error
        ) {
            controller.dismiss(animated: true)
            print("Scan failed: \(error.localizedDescription)")
        }
    }
}
```

## Uploading Pages to R2 via Pre-signed URL

```swift
// OCRUploader.swift
import Foundation
import UIKit

actor OCRUploader {
    private let workerBase = URL(string: "https://api.example.com")!

    /// Obtain a pre-signed R2 PUT URL from the Workers API.
    func presignUpload(pageIndex: Int, accessToken: String) async throws -> (objectKey: String, uploadURL: URL) {
        var req = URLRequest(url: workerBase.appendingPathComponent("/scan/presign"))
        req.httpMethod = "POST"
        req.setValue("Bearer \(accessToken)", forHTTPHeaderField: "Authorization")
        req.setValue("application/json", forHTTPHeaderField: "Content-Type")
        req.httpBody = try JSONEncoder().encode(["pageIndex": pageIndex, "contentType": "image/jpeg"])

        let (data, _) = try await URLSession.shared.data(for: req)
        let body      = try JSONDecoder().decode(PresignResponse.self, from: data)
        return (body.objectKey, URL(string: body.uploadURL)!)
    }

    /// PUT the JPEG bytes directly to R2 using the pre-signed URL.
    func uploadJPEG(_ image: CGImage, to uploadURL: URL) async throws {
        guard let jpeg = UIImage(cgImage: image).jpegData(compressionQuality: 0.82) else {
            throw OCRError.compressionFailed
        }
        var req         = URLRequest(url: uploadURL)
        req.httpMethod  = "PUT"
        req.setValue("image/jpeg",                 forHTTPHeaderField: "Content-Type")
        req.setValue(String(jpeg.count),            forHTTPHeaderField: "Content-Length")
        req.httpBody    = jpeg

        let (_, response) = try await URLSession.shared.data(for: req)
        guard (response as? HTTPURLResponse)?.statusCode == 200 else {
            throw OCRError.uploadFailed
        }
    }
}

struct PresignResponse: Decodable { let objectKey: String; let uploadURL: String }
enum OCRError: Error { case compressionFailed; case uploadFailed; case extractionFailed }
```

## Requesting OCR Extraction

```swift
// OCRClient.swift
struct ExtractedDocument: Decodable {
    let vendor: String?
    let total: String?
    let date: String?
    let lineItems: [LineItem]?
    let rawText: String

    struct LineItem: Decodable { let description: String; let amount: String }
}

func extractDocument(objectKey: String, accessToken: String) async throws -> ExtractedDocument {
    let base = URL(string: "https://api.example.com")!
    var req  = URLRequest(url: base.appendingPathComponent("/scan/extract"))
    req.httpMethod = "POST"
    req.setValue("Bearer \(accessToken)", forHTTPHeaderField: "Authorization")
    req.setValue("application/json",       forHTTPHeaderField: "Content-Type")
    req.httpBody   = try JSONEncoder().encode(["objectKey": objectKey])

    let (data, response) = try await URLSession.shared.data(for: req)
    guard (response as? HTTPURLResponse)?.statusCode == 200 else {
        throw OCRError.extractionFailed
    }
    return try JSONDecoder().decode(ExtractedDocument.self, from: data)
}
```

## Cloudflare Workers Backend

```typescript
// worker.ts
import { Ai } from '@cloudflare/ai'

// POST /scan/presign  –  generate a signed R2 PUT URL
async function handlePresign(request: Request, env: Env): Promise<Response> {
  const userId = await authenticate(request, env)
  if (!userId) return new Response('Unauthorized', { status: 401 })

  const { pageIndex } = await request.json<{ pageIndex: number }>()
  const objectKey     = `scans/${userId}/${Date.now()}-page${pageIndex}.jpg`

  // R2 presigned URL (Workers R2 binding presign API)
  const url = await env.SCAN_BUCKET.createMultipartUpload(objectKey)
  // For single-part: use a signed URL via a separate signing Worker or AWS-compat SDK
  // Here we use a direct upload approach via a Workers pre-signed path
  const signedUrl = `https://uploads.example.com/${objectKey}?token=${await signKey(objectKey, env)}`

  return Response.json({ objectKey, uploadURL: signedUrl })
}

// POST /scan/extract  –  read image from R2, run Workers AI, return JSON
async function handleExtract(request: Request, env: Env): Promise<Response> {
  const userId = await authenticate(request, env)
  if (!userId) return new Response('Unauthorized', { status: 401 })

  const { objectKey } = await request.json<{ objectKey: string }>()

  if (!objectKey.startsWith(`scans/${userId}/`)) {
    return new Response('Forbidden', { status: 403 })
  }

  const object = await env.SCAN_BUCKET.get(objectKey)
  if (!object) return new Response('Not Found', { status: 404 })

  const imageBytes = new Uint8Array(await object.arrayBuffer())

  const ai     = new Ai(env.AI)
  const result = await ai.run('@cf/meta/llama-3.2-11b-vision-instruct', {
    image: Array.from(imageBytes),
    prompt: `You are a document parser. Extract from this receipt or invoice:
- vendor name
- total amount
- date
- line items (description, amount)
Return ONLY valid JSON with keys: vendor, total, date, lineItems (array of {description, amount}), rawText.`,
    max_tokens: 1024,
  })

  let parsed: Record<string, unknown>
  try {
    const jsonMatch = (result as { response: string }).response.match(/\{[\s\S]*\}/)
    parsed = jsonMatch ? JSON.parse(jsonMatch[0]) : {}
  } catch {
    parsed = {}
  }

  // Clean up R2 object after extraction
  await env.SCAN_BUCKET.delete(objectKey)

  return Response.json({
    vendor:    parsed.vendor    ?? null,
    total:     parsed.total     ?? null,
    date:      parsed.date      ?? null,
    lineItems: parsed.lineItems ?? [],
    rawText:   parsed.rawText   ?? '',
  })
}

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    const url = new URL(request.url)
    if (url.pathname === '/scan/presign') return handlePresign(request, env)
    if (url.pathname === '/scan/extract') return handleExtract(request, env)
    return new Response('Not Found', { status: 404 })
  },
}
```

## Anti-patterns

- **Sending the raw image in the request body to the Worker** — Workers have a 100 MB request
  body limit and subrequest limits. Use R2 as the image bus: upload separately, pass the key.
- **Running `VNRecognizeTextRequest` on the main thread** — Vision requests are CPU-bound; always
  dispatch to a background `DispatchQueue` or use Swift concurrency with `Task.detached`.
- **Storing scanned images in the app's Documents directory indefinitely** — delete the local
  JPEG after confirming the R2 upload succeeded; delete the R2 object after extraction.
- **Skipping the userId ownership check before R2 access** — always verify `objectKey` starts
  with the authenticated user's prefix to prevent horizontal privilege escalation.

## Gotchas

- `VNDocumentCameraViewController` requires camera permission (`NSCameraUsageDescription`). Check
  `AVCaptureDevice.authorizationStatus` before presenting; the system will not prompt twice if
  the user previously denied.
- Workers AI vision models have a context window for images. For long multi-page documents,
  send pages individually and merge the JSON responses on the client.
- `@cf/meta/llama-3.2-11b-vision-instruct` is a beta binding as of 2026-08; check the Workers
  AI model catalogue for GA status and substitutes.
- The pre-signed R2 URL approach shown is a simplified pattern. For production, implement full
  HMAC-signed URL generation using the `crypto.subtle` API inside a dedicated signing Worker.
