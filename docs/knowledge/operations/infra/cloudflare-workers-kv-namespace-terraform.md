# Cloudflare Workers KV Namespace Terraform Management

Date: 2026-08-23
Author: example.com
Status: production

---

## Symptom / Use-case

You provision Workers KV namespaces by hand through the dashboard or wrangler CLI. Namespaces drift between environments, bindings are missing from `wrangler.toml`, and there is no audit trail of who created what. You need reproducible, reviewable KV infrastructure tied to your GitOps pipeline.

---

## Context

Cloudflare Workers KV is a globally distributed key-value store. Each namespace is a logical bucket. Namespaces must be created at the account level, then bound to individual Workers. Terraform's `cloudflare` provider (v4+) manages namespaces via `cloudflare_workers_kv_namespace` and can write values via `cloudflare_workers_kv`. The binding between a namespace and a Worker is expressed in `wrangler.toml` using the namespace ID output from Terraform.

Common patterns:
- One namespace per environment per logical dataset (e.g. `config-staging`, `config-production`)
- A shared read-only namespace for static reference data
- Per-tenant namespaces in a Workers for Platforms setup

## Namespace Provisioning

```hcl
# terraform/kv.tf

terraform {
  required_providers {
    cloudflare = {
      source  = "cloudflare/cloudflare"
      version = "~> 4.0"
    }
  }
}

variable "account_id" {
  type = string
}

locals {
  environments = ["staging", "production"]
  datasets     = ["config", "sessions", "feature-flags"]
}

resource "cloudflare_workers_kv_namespace" "app" {
  for_each   = toset([
    for pair in setproduct(local.environments, local.datasets) :
    "${pair[0]}-${pair[1]}"
  ])
  account_id = var.account_id
  title      = "app-${each.key}"
}

output "kv_namespace_ids" {
  value = {
    for k, ns in cloudflare_workers_kv_namespace.app : k => ns.id
  }
}
```

## Seeding Static Values via Terraform

Terraform can write KV entries. Limit this to immutable bootstrap data — session data belongs in application code.

```hcl
resource "cloudflare_workers_kv" "feature_flags_defaults" {
  account_id   = var.account_id
  namespace_id = cloudflare_workers_kv_namespace.app["production-feature-flags"].id
  key          = "defaults"
  value = jsonencode({
    newCheckout  = false
    betaDashboard = false
    aiSuggestions = true
  })
}
```

## Wrangler Binding via Terraform Output

Generate `wrangler.toml` bindings from Terraform outputs using a `local_file` resource or a CI pipeline step.

```hcl
# Write a partial wrangler config fragment consumed by CI
resource "local_file" "wrangler_kv_bindings" {
  filename = "${path.module}/../wrangler-kv-bindings.toml"
  content  = <<-EOT
    # Auto-generated by Terraform — do not edit by hand
    [[kv_namespaces]]
    binding = "CONFIG"
    id      = "${cloudflare_workers_kv_namespace.app["production-config"].id}"

    [[kv_namespaces]]
    binding = "FEATURE_FLAGS"
    id      = "${cloudflare_workers_kv_namespace.app["production-feature-flags"].id}"

    [[kv_namespaces]]
    binding = "SESSIONS"
    id      = "${cloudflare_workers_kv_namespace.app["production-sessions"].id}"
  EOT
}
```

CI merges this fragment into `wrangler.toml` before `wrangler deploy`.

## Workers TypeScript Usage

```typescript
// src/index.ts
export interface Env {
  CONFIG: KVNamespace;
  FEATURE_FLAGS: KVNamespace;
  SESSIONS: KVNamespace;
}

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    // Read with cache TTL to reduce per-request KV reads
    const flags = await env.FEATURE_FLAGS.get("defaults", {
      type: "json",
      cacheTtl: 300,
    }) as Record<string, boolean> | null;

    if (flags?.newCheckout) {
      return new Response("new checkout flow");
    }

    return new Response("standard checkout flow");
  },
};
```

## Bulk Migration Script (TypeScript)

When moving namespaces between accounts, use the Workers API directly:

```typescript
// scripts/migrate-kv.ts
const SOURCE_ACCOUNT = process.env.SOURCE_ACCOUNT_ID!;
const DEST_ACCOUNT   = process.env.DEST_ACCOUNT_ID!;
const API_TOKEN      = process.env.CF_API_TOKEN!;

async function listKeys(accountId: string, namespaceId: string): Promise<string[]> {
  const url = `https://api.cloudflare.com/client/v4/accounts/${accountId}/storage/kv/namespaces/${namespaceId}/keys`;
  const res = await fetch(url, {
    headers: { Authorization: `Bearer ${API_TOKEN}` },
  });
  const data = await res.json() as { result: Array<{ name: string }> };
  return data.result.map((k) => k.name);
}

async function getValue(accountId: string, namespaceId: string, key: string): Promise<string> {
  const url = `https://api.cloudflare.com/client/v4/accounts/${accountId}/storage/kv/namespaces/${namespaceId}/values/${key}`;
  const res = await fetch(url, { headers: { Authorization: `Bearer ${API_TOKEN}` } });
  return res.text();
}

async function putValue(accountId: string, namespaceId: string, key: string, value: string): Promise<void> {
  const url = `https://api.cloudflare.com/client/v4/accounts/${accountId}/storage/kv/namespaces/${namespaceId}/values/${key}`;
  await fetch(url, {
    method: "PUT",
    headers: { Authorization: `Bearer ${API_TOKEN}`, "Content-Type": "text/plain" },
    body: value,
  });
}

// Usage: node -r tsx/register scripts/migrate-kv.ts <src-ns-id> <dest-ns-id>
```

## Anti-patterns

- **Creating namespaces manually then importing**: import state immediately after creation or Terraform will create a duplicate namespace.
- **Storing session data in KV with Terraform**: KV values managed by Terraform are in state — session keys contain PII and must never touch state files.
- **One namespace for all environments**: namespace isolation prevents a staging `flushAll` from wiping production data.
- **Omitting `cacheTtl` on hot-path reads**: without a cache TTL every Worker request is a separate KV read; at scale this multiplies costs and latency.
- **Hardcoding namespace IDs in source code**: always use binding names (`env.CONFIG`) not raw IDs.

---

## Gotchas

- KV namespace IDs are UUIDs generated at creation time. They cannot be chosen. Store them in Terraform remote state and reference via `terraform_remote_state` or SSM.
- The `cloudflare_workers_kv` resource writes are synchronous but propagation to all PoPs takes up to 60 seconds — do not test immediately after `terraform apply` in CI.
- Terraform destroy on a namespace will delete all keys inside it with no confirmation prompt. Guard with `prevent_destroy = true` in lifecycle blocks for production.
- The Cloudflare provider's `cloudflare_workers_kv_namespace` resource title is not unique-enforced — Terraform apply on a different workspace creates a second namespace with the same title.
- Free plan accounts are limited to 1 namespace; pay-as-you-go allows up to 100.

## Verification

```bash
# List all namespaces in account
curl -s "https://api.cloudflare.com/client/v4/accounts/$CF_ACCOUNT_ID/storage/kv/namespaces" \
  -H "Authorization: Bearer $CF_API_TOKEN" | jq '.result[] | {id, title}'

# Check a specific key exists
curl -s "https://api.cloudflare.com/client/v4/accounts/$CF_ACCOUNT_ID/storage/kv/namespaces/$NS_ID/values/defaults" \
  -H "Authorization: Bearer $CF_API_TOKEN"

# Confirm Terraform state matches live resources
terraform plan -var="account_id=$CF_ACCOUNT_ID" 2>&1 | grep -E "(No changes|namespace)"
```
